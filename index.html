function makeShowCard(show,showId){ const card=document.createElement("article"); card.className="show-card"; const poster=approvedPosters[showId]; const image=document.createElement("img"); image.className="show-image"; image.alt=(show.name||"Singing show")+" poster"; image.loading="lazy"; const approvedImage=safePosterUrl(poster&&poster.imageUrl); image.src=approvedImage|| "https://images.unsplash.com/photo-1501386761578-eac5c94b800a?auto=format&fit=crop&w=800&q=85"; image.onerror=()=>{ image.onerror=null; image.src="https://images.unsplash.com/photo-1506157786151-b8491531f063?auto=format&fit=crop&w=800&q=80"; }; const info=document.createElement("div"); info.className="show-info"; const audienceFee=Number(show.audienceFee??show.audienceEntryFee??show.entryFee??0); const performerFee=Number(show.performerFee??show.performerEntryFee??0); const audiencePaid=Number.isFinite(audienceFee)&&audienceFee>0; const performerPaid=Number.isFinite(performerFee)&&performerFee>0; const label=document.createElement("span"); label.className="show-label"; label.textContent=audiencePaid?"PAID ENTRY SHOW":"FREE ENTRY SHOW"; const title=document.createElement("h3"); title.textContent=show.name||show.title||"Untitled Show"; const host=document.createElement("p"); host.textContent="Host: "+(show.hostName||"Approved Host"); const dateTime=document.createElement("p"); dateTime.textContent="Date: "+(show.date||"Not specified")+ " | Time: "+(show.time||"Not specified"); const description=document.createElement("p"); description.textContent=show.description||"Join this live musical experience."; const entry=document.createElement("p"); entry.className="show-entry"; entry.textContent=audiencePaid ?"Audience Entry Fee: ₹"+audienceFee :"Audience Entry: FREE"; const performerEntry=document.createElement("p"); performerEntry.className="show-entry"; performerEntry.textContent=performerPaid ?"Performer Registration Fee: ₹"+performerFee :"Performer Registration: FREE"; const performerButton=document.createElement("a"); performerButton.className="show-button"; performerButton.textContent="REGISTER AS PERFORMER"; performerButton.href="singer-registration.html?showId="+encodeURIComponent(showId); const audienceButton=document.createElement("a"); audienceButton.className="show-button secondary"; if(!audiencePaid){ audienceButton.textContent="JOIN FREE AS AUDIENCE"; audienceButton.href="audience-entry.html?showId="+encodeURIComponent(showId); }else{ const hostNumber=getHostWhatsApp(show); if(hostNumber){ audienceButton.textContent="PAY & JOIN SHOW"; const message="Hello, I want to join your paid show: "+ (show.name||show.title||"Singing Show")+ ". Please share payment details and audience entry instructions."; audienceButton.href="https://wa.me/"+hostNumber+ "?text="+encodeURIComponent(message); audienceButton.target="_blank"; audienceButton.rel="noopener noreferrer"; }else{ audienceButton.textContent="CONTACT HOST FOR ENTRY"; audienceButton.classList.add("disabled"); audienceButton.removeAttribute("href"); } } info.append( label, title, host, dateTime, description, entry, performerEntry, performerButton, audienceButton ); card.append(image,info); return card; }

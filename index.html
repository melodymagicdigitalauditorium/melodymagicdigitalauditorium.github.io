function makeShowCard(show,showId){ const card=document.createElement("article"); card.className="show-card"; const poster=approvedPosters[showId]; const image=document.createElement("img"); image.className="show-image"; image.alt=(show.name||"Singing show")+" poster"; image.loading="lazy"; const approvedImage=safePosterUrl(poster&&poster.imageUrl); image.src=approvedImage|| "https://images.unsplash.com/photo-1501386761578-eac5c94b800a?auto=format&fit=crop&w=800&q=85"; image.onerror=()=>{ image.onerror=null; image.src="https://images.unsplash.com/photo-1506157786151-b8491531f063?auto=format&fit=crop&w=800&q=80"; }; const info=document.createElement("div"); info.className="show-info"; // Fee display: zero = FREE, null/blank = Not announced. function formatFee(value){ if(value===null||value===undefined||String(value).trim()===""){ return "Not announced"; } const amount=Number(value); if(!Number.isFinite(amount)||amount<0){ return "Not announced"; } return amount===0?"FREE":"₹"+amount; } // Read the exact Firebase Host Dashboard fields. const soloFee=formatFee(show.soloEntryFee); const duetFee=formatFee(show.duetEntryFee); const audienceFee=formatFee(show.audienceFee); const audiencePaid=audienceFee!=="FREE" && audienceFee!=="Not announced"; const label=document.createElement("span"); label.className="show-label"; label.textContent=audiencePaid ?"PAID ENTRY SHOW" :"FREE ENTRY SHOW"; const title=document.createElement("h3"); title.textContent=show.name||show.title||"Untitled Show"; const host=document.createElement("p"); host.textContent="Host: "+(show.hostName||"Approved Host"); const dateTime=document.createElement("p"); dateTime.textContent= "Date: "+(show.date||"Not specified")+ " | Time: "+(show.time||"Not specified"); const description=document.createElement("p"); description.textContent= show.description||"Join this live musical experience."; // Separate fee details. const solo=document.createElement("p"); solo.className="performer-entry"; solo.textContent="🎤 Solo Performer Fee: "+soloFee; const duet=document.createElement("p"); duet.className="performer-entry"; duet.textContent="🎤 Duet Performer Fee: "+duetFee; const audience=document.createElement("p"); audience.className="show-entry"; audience.textContent="🎧 Audience Entry Fee: "+audienceFee; const performerButton=document.createElement("a"); performerButton.className="show-button"; performerButton.textContent="REGISTER AS PERFORMER"; performerButton.href= "singer-registration.html?showId="+encodeURIComponent(showId); const audienceButton=document.createElement("a"); audienceButton.className="show-button secondary"; audienceButton.textContent="REGISTER AS AUDIENCE"; audienceButton.href= "audience-entry.html?showId="+encodeURIComponent(showId); info.append( label, title, host, dateTime, description, solo, duet, audience, performerButton, audienceButton ); card.append(image,info); return card; }

function renderRegistrations(showId,data){ const box=$("registrationsList"); const entries=Object.entries(data||{}); if(!entries.length){ box.innerHTML='
No registrations yet.
'; return; } box.innerHTML=entries.map(([id,p])=>`

${escapeHTML(p.singerName||p.name||"Performer")}

Mobile: ${escapeHTML(p.mobile||p.phone||"-")}

Song: ${escapeHTML(p.songName||p.song||"-")}

Movie: ${escapeHTML(p.movieName||p.movie||"-")}

Performance Type: ${escapeHTML(p.performanceType||p.type||"-")}

Status: ${escapeHTML(p.status||"Pending")} Payment: ${escapeHTML(p.paymentStatus||"Unpaid")}

`).join(""); // APPROVE: automatically mark payment as Paid box.querySelectorAll("[data-approve]").forEach(b=>{ b.onclick=async()=>{ const id=b.dataset.approve; if(!confirm("Confirm payment and approve this performer?"))return; b.disabled=true; b.textContent="Approving..."; try{ const snap=await get(ref(db,`showRegistrations/${showId}/${id}`)); if(!snap.exists()){ throw new Error("Registration not found."); } const performer=snap.val(); if(!performer.singerUid){ throw new Error("Performer login UID is missing. Ask the performer to register again."); } await updatePerformer(showId,id,{ status:"Approved", paymentStatus:"Paid", approvedAt:Date.now() }); await addRegistrationToQueue(showId,id); alert("Payment confirmed! Performer approved and added to queue."); }catch(e){ alert("Approval failed: "+e.message); }finally{ b.disabled=false; b.textContent="APPROVE & ADD TO QUEUE"; } }; }); // REJECT box.querySelectorAll("[data-reject]").forEach(b=>{ b.onclick=async()=>{ const id=b.dataset.reject; if(!confirm("Reject this performer registration?"))return; b.disabled=true; try{ await updatePerformer(showId,id,{ status:"Rejected" }); await remove(ref(db,`performanceQueue/${showId}/${id}`)); alert("Registration rejected."); }catch(e){ alert("Rejection failed: "+e.message); }finally{ b.disabled=false; } }; }); }

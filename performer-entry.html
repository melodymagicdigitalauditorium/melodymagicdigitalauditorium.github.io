async function findExistingRegistration(){ const q=query( ref(db,"showRegistrations/"+showId), orderByChild("singerUid"), equalTo(currentUser.uid) ); const snap=await get(q); if(!snap.exists())return false; const entries=Object.entries(snap.val()); // Prefer an approved registration over a pending duplicate. const priority={ "approved":1, "queued":2, "waitingqueue":3, "pending":4, "completed":5, "rejected":6 }; entries.sort((a,b)=>{ const sa=String(a[1].status||"Pending").trim().toLowerCase(); const sb=String(b[1].status||"Pending").trim().toLowerCase(); const pa=priority[sa]||99; const pb=priority[sb]||99; if(pa!==pb)return pa-pb; return Number(b[1].createdAt||0)- Number(a[1].createdAt||0); }); const valid=entries.find(([id,p])=> String(p.status||"Pending").trim().toLowerCase() !=="rejected" ); if(!valid)return false; registrationId=valid[0]; renderRegistration(valid[1]); listenForApproval(); return true; }
